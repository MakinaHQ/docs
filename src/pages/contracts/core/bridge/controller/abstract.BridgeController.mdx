        mapping(uint16 bridgeId => uint256 maxBridgeLossBps) _maxBridgeLossBps;
        mapping(uint16 bridgeId => bool isOutTransferEnabled) _isOutTransferEnabled;
        mapping(address addr => bool isAdapter) _isBridgeAdapter;
    }
